УЛУЧШЕННАЯ ПРОВЕРКА ДУБЛИКАТОВ (12 ЧАСОВ) --- def clean_line(text: str) -> str: # Удаляем спецсимволы, пунктуацию и приведи к нижнему регистру return re.sub(r'[^\w\s]', '', text).strip().lower() def is_duplicate_vacancy(raw_text: str) -> bool: now = time.time() global sent_vacancies_cache # Окно очистки: 12 часов (43 200 секунд) sent_vacancies_cache = [v for v in sent_vacancies_cache if now - v["timestamp"] <= 43200] raw_lines = [clean_line(line) for line in raw_text.split('\n')] new_lines = [line for line in raw_lines if len(line) > 5] if not new_lines: return False new_lines_set = set(new_lines) total_lines = len(new_lines_set) for item in sent_vacancies_cache: matching_lines = len(new_lines_set.intersection(item["lines"])) if total_lines >= 3 and matching_lines >= 3: return True elif total_lines < 3 and matching_lines == total_lines: return True return False def register_sent_vacancy(raw_text: str): raw_lines = [clean_line(line) for line in raw_text.split('\n')] lines = set([line for line in raw_lines if len(line) > 5]) sent_vacancies_cache.append({ "lines": lines, "timestamp": time.time() }) # --- 3. JSON ИСТОРИЯ И ДАТАСЕТ --- def load_json(filepath, default): if os.path.exists(filepath): try: with open(filepath, 'r', encoding='utf-8') as f: return json.load(f) except Exception as e: print(f"[-] Ошибка чтения {filepath}: {e}") return default def save_json(filepath, data): try: with open(filepath, 'w', encoding='utf-8') as f: json.dump(data, f, ensure_ascii=False, indent=2) except Exception as e: print(f"[-] Ошибка записи {filepath}: {e}") # --- 4. ИИ-АНАЛИЗ (Gemini API с подробными логами) --- async def evaluate_with_ai(text: str) -> bool: if not GEMINI_API_KEY: print("[!] GEMINI_API_KEY отсутствует в переменных окружения. ИИ пропущен.") return True print(f"[AI] Отправляем текст в Gemini API (длина: {len(text)} симв.)...") dataset = load_json(DATASET_FILE, {"likes": [], "dislikes": []}) good_examples = dataset.get("likes", [])[-5:] bad_examples = dataset.get("dislikes", [])[-5:] prompt = f""" Ты — фильтр вакансий для видеомонтажера (Reels, Shorts, TikTok). Оцени текст и ответь, является ли он РЕАЛЬНОЙ ВАКАНСИЕЙ (работодатель ищет исполнителя). ПРИМЕРЫ ХОРОШИХ ВАКАНСИЙ (Одобрены): {json.dumps(good_examples, ensure_ascii=False)} ПРИМЕРЫ МУСОРА ИЛИ РЕЗЮМЕ (Отклонены): {json.dumps(bad_examples, ensure_ascii=False)} Текст для оценки: "{text}" Ответь СТРОГО одним словом: YES (если это вакансия) или NO (если это резюме, услуги, SMM или мусор). """ # Модель gemini-1.5-flash с прямой поддержкой REST v1beta url = f"https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key={GEMINI_API_KEY}" payload = {"contents": [{"parts": [{"text": prompt}]}]} try: async with ClientSession() as session: async with session.post(url, json=payload, timeout=10) as resp: if resp.status == 200: data = await resp.json() answer = data['candidates'][0]['content']['parts'][0]['text'].strip().upper() print(f"[AI] Ответ от Gemini: {answer}") return 'YES' in answer else: err_body = await resp.text() print(f"[-] Ошибка ответа Gemini API (Status {resp.status}): {err_body}") except Exception as e: print(f"[-] Ошибка соединения с Gemini API: {e}") return True # --- 5. TELEGRAM BOT API --- async def send_to_bot(text: str) -> int | None: url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendMessage" payload = { "chat_id": MY_CHAT_ID, "text": text, "parse_mode": "HTML", "disable_web_page_preview": True, "reply_markup": { "inline_keyboard": [[ {"text": "👍 Подходит", "callback_data": "vote_like"}, {"text": "👎 Мусор", "callback_data": "vote_dislike"} ]] } } async with ClientSession() as session: async with session.post(url, json=payload) as resp: if resp.status == 200: data = await resp.json() return data.get("result", {}).get("message_id") else: print(f"[-] Ошибка отправки: {await resp.text()}") return None async def delete_bot_message(msg_id: int): url = f"https://api.telegram.org/bot{BOT_TOKEN}/deleteMessage" payload = {"chat_id": MY_CHAT_ID, "message_id": msg_id} async with ClientSession() as session: await session.post(url, json=payload) async def edit_bot_message_text(msg_id: int, text: str): url = f"https://api.telegram.org/bot{BOT_TOKEN}/editMessageText" payload = { "chat_id": MY_CHAT_ID, "message_id": msg_id, "text": text, "parse_mode": "HTML", "disable_web_page_preview": True } async with ClientSession() as session: await session.post(url, json=payload) async def answer_callback_query(callback_id: str, text: str): url = f"https://api.telegram.org/bot{BOT_TOKEN}/answerCallbackQuery" payload = {"callback_query_id": callback_id, "text": text} async with ClientSession() as session: await session.post(url, json=payload) # --- 6. ОБРАБОТКА КНОПОК 👍 / 👎 --- async def bot_updates_loop(): offset = 0 url = f"https://api.telegram.org/bot{BOT_TOKEN}/getUpdates" while True: try: async with ClientSession() as session: async with session.get(url, params={"offset": offset, "timeout": 20}) as resp: if resp.status == 200: data = await resp.json() for update in data.get("result", []): offset = update["update_id"] + 1 if "callback_query" in update: cb = update["callback_query"] cb_id = cb["id"] data_cb = cb.get("data") msg = cb.get("message", {}) msg_id = msg.get("message_id") raw_text = msg.get("text", "") dataset = load_json(DATASET_FILE, {"likes": [], "dislikes": []}) if data_cb == "vote_like": dataset["likes"].append(raw_text) save_json(DATASET_FILE, dataset) await answer_callback_query(cb_id, "Записано: Подходит 👍") await edit_bot_message_text(msg_id, f"✅ [ОДОБРЕНО]\n\n{html.escape(raw_text)}") elif data_cb == "vote_dislike": dataset["dislikes"].append(raw_text) save_json(DATASET_FILE, dataset) await answer_callback_query(cb_id, "Записано в мусор 👎") await delete_bot_message(msg_id) history = load_json(HISTORY_FILE, []) history = [h for h in history if h.get("msg_id") != msg_id] save_json(HISTORY_FILE, history) except Exception as e: print(f"[-] Ошибка в опросе кнопок бота: {e}") await asyncio.sleep(2) # --- 7. ФОНОВЫЕ ЗАДАЧИ --- async def auto_delete_old_messages_loop(): THREE_DAYS_SEC = 3 * 24 * 3600 while True: await asyncio.sleep(600) try: history = load_json(HISTORY_FILE, []) if not history: continue now = time.time() updated_history = [] for item in history: msg_id = item.get("msg_id") ts = item.get("timestamp", 0) if now - ts >= THREE_DAYS_SEC: if msg_id: await delete_bot_message(msg_id) print(f"[x] Сообщение {msg_id} удалено (3 дня).") else: updated_history.append(item) if len(updated_history) != len(history): save_json(HISTORY_FILE, updated_history) except Exception as e: print(f"[-] Ошибка автоудаления: {e}") async def cleanup_recent_posts_loop(): while True: await asyncio.sleep(120) now = time.time() to_del = [k for k, v in recent_channel_posts.items() if now - v["timestamp"] > 300] for k in to_del: recent_channel_posts.pop(k, None) # --- 8. ВЕБ-СЕРВЕР --- async def handle_ping(request): return web.Response(text="Бот активен") async def start_web_server(): app = web.Application() app.router.add_get('/', handle_ping) runner = web.AppRunner(app) await runner.setup() port = int(os.environ.get("PORT", 8080)) site = web.TCPSite(runner, '0.0.0.0', port) await site.start() print(f"[+] Веб-сервер запущен на порту {port}") # --- 9. ОБРАБОТКА ВАКАНСИЙ И СЕРВЕРА TELEGRAM --- @client.on(events.MessageDeleted()) async def handle_deleted_message(event): now = time.time() chat_id = getattr(event, 'chat_id', None) for deleted_id in event.deleted_ids: matched_keys = [] for (c_id, m_id) in list(recent_channel_posts.keys()): if m_id == deleted_id: if chat_id is None or c_id == chat_id: matched_keys.append((c_id, m_id)) for key in matched_keys: post_info = recent_channel_posts.pop(key, None) if post_info: elapsed = now - post_info["timestamp"] bot_msg_id = post_info["bot_msg_id"] if elapsed <= 60: await delete_bot_message(bot_msg_id) print(f"[x] Вакансия удалена из источника за 1 мин. Сообщение {bot_msg_id} удалено.") history = load_json(HISTORY_FILE, []) updated_history = [h for h in history if h.get("msg_id") != bot_msg_id] if len(updated_history) != len(history): save_json(HISTORY_FILE, updated_history) @client.on(events.NewMessage(chats=TARGET_CHATS)) async def handle_new_message(event): message_text = event.raw_text.lower() # 1. Фильтр стоп-слов if any(ex in message_text for ex in EXCLUDE_KEYWORDS): print(f"[x] Заблокировано стоп-словом из чата {event.chat_id}") return # 2. Фильтр дубликатов (12 часов) if is_duplicate_vacancy(event.raw_text): print(f"[x] Пропущен дубликат вакансии из чата {event.chat_id}") return # 3. Фильтр через ИИ Gemini is_valid_by_ai = await evaluate_with_ai(event.raw_text) if not is_valid_by_ai: print("[x] Gemini ИИ отклонил сообщение как нерелевантное.") return chat = await event.get_chat() chat_title = getattr(chat, 'title', 'Чат/Канал') if getattr(chat, 'username', None): message_link = f"https://t.me/{chat.username}/{event.id}" else: message_link = "Ссылка недоступна (приватный чат)" keywords_str = ', '.join(set(found_keywords)) alert_text = ( "🎯 НАЙДЕНА ВАКАНСИЯ ПО МОНТАЖУ!\n\n" f"📌 Источник: {html.escape(chat_title)}\n" f"🔑 Ключевые слова: {html.escape(keywords_str)}\n" f"🔗 Ссылка: {message_link}\n\n" f"📝 Текст объявления:\n{html.escape(event.raw_text)}" ) bot_msg_id = await send_to_bot(alert_text) if bot_msg_id: print(f"[+] Вакансия переслана из: {chat_title}") now = time.time() history = load_json(HISTORY_FILE, []) history.append({"msg_id": bot_msg_id, "timestamp": now}) save_json(HISTORY_FILE, history) recent_channel_posts[(event.chat_id, event.id)] = { "bot_msg_id": bot_msg_id, "timestamp": now, "text": event.raw_text } # --- 10.
